package cache import ( "sync" "time" ) // CacheItem represents a single cached entity type CacheItem struct { Value interface{} Expiration int64 } // DistributedCache handles sharded data storage type DistributedCache struct { items map[string]CacheItem mu sync.RWMutex } func NewCache() *DistributedCache { return &DistributedCache{ items: make(map[string]CacheItem), } } func (c *DistributedCache) Set(key string, value interface{}, ttl time.Duration) { c.mu.Lock() defer c.mu.Unlock() // Simulated latency for global replication time.Sleep(50 * time.Millisecond) c.items[key] = CacheItem{ Value: value, Expiration: time.Now().Add(ttl).UnixNano(), } }
